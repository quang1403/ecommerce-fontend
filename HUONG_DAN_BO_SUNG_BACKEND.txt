// comments.controller.js hoặc tương tự

exports.getCommentsByProduct = async (req, res) => {
  try {
    const { productId } = req.params;
    
    // Lấy danh sách comments và populate thông tin user
    const comments = await Comment.find({ productId })
      .populate('userId', 'fullName name email')
      .sort({ createdAt: -1 })
      .lean();
    
    // Thêm thông tin hasPurchased cho mỗi comment
    // ⚠️ VÌ BẮT BUỘC ĐĂNG NHẬP MỚI COMMENT, NÊN TẤT CẢ ĐỀU CÓ userId
    const commentsWithPurchaseStatus = await Promise.all(
      comments.map(async (comment) => {
        // Kiểm tra user này có đơn hàng hoàn thành với sản phẩm này không
        const purchaseExists = await Order.exists({
          customerId: comment.userId._id || comment.userId,
          status: { $in: [3, 4] }, // 3 = Đã giao, 4 = Hoàn thành (tùy theo status code của bạn)
          'items.productId': productId
        });
        
        return {
          ...comment,
          hasPurchased: !!purchaseExists
        };
      })
    );
    
    res.status(200).json({
      success: true,
      data: commentsWithPurchaseStatus
    });
    
  } catch (error) {
    res.status(500).json({
      success: false,
      message: 'Lỗi khi lấy danh sách bình luận',
      error: error.message
    });
  }
};

CÁCH 1: Sử dụng aggregation pipeline
-------------------------------------

exports.getCommentsByProduct = async (req, res) => {
  try {
    const { productId } = req.params;
    
    const comments = await Comment.aggregate([
      // Match comments của sản phẩm
      { $match: { productId: mongoose.Types.ObjectId(productId) } },
      
      // Lookup thông tin user
      {
        $lookup: {
          from: 'users',
          localField: 'userId',
          foreignField: '_id',
          as: 'user'
        }
      },
      { $unwind: { path: '$user', preserveNullAndEmptyArrays: true } },
      
      // Lookup kiểm tra đơn hàng
      {
        $lookup: {
          from: 'orders',
          let: { userId: '$userId', prodId: productId },
          pipeline: [
            {
              $match: {
                $expr: {
                  $and: [
                    { $eq: ['$customerId', '$$userId'] },
                    { $in: ['$status', [3, 4]] },
                    {
                      $gt: [
                        {
                          $size: {
                            $filter: {
                              input: '$items',
                              as: 'item',
                              cond: { $eq: ['$$item.productId', '$$prodId'] }
                            }
                          }
                        },
                        0
                      ]
                    }
                  ]
                }
              }
            },
            { $limit: 1 }
          ],
          as: 'purchases'
        }
      },
      
      // Thêm field hasPurchased
      {
        $addFields: {
          hasPurchased: { $gt: [{ $size: '$purchases' }, 0] }
        }
      },
      
      // Loại bỏ field không cần thiết
      { $project: { purchases: 0 } },
      
      // Sắp xếp
      { $sort: { createdAt: -1 } }
    ]);
    
    res.status(200).json({
      success: true,
      data: comments
    });
    
  } catch (error) {
    res.status(500).json({
      success: false,
      message: 'Lỗi khi lấy danh sách bình luận'
    });
  }
};


CÁCH 2: Thêm field hasPurchased trực tiếp vào Comment Model

Khi user tạo comment, kiểm tra và lưu luôn:

// Khi tạo comment mới
exports.createComment = async (req, res) => {
  try {
    const { productId } = req.params;
    const userId = req.user._id; // Từ middleware auth
    
    // Kiểm tra user đã mua sản phẩm này chưa
    const hasPurchased = await Order.exists({
      customerId: userId,
      status: { $in: [3, 4] },
      'items.productId': productId
    });
    
    const comment = await Comment.create({
      ...req.body,
      productId,
      userId,
      hasPurchased: !!hasPurchased
    });
    
    res.status(201).json({
      success: true,
      data: comment
    });
    
  } catch (error) {
    res.status(500).json({
      success: false,
      message: 'Lỗi khi tạo bình luận'
    });
  }
};

BƯỚC 4: MAPPING STATUS ĐỐN HÀNG
================================================================

Cần xác định status code nào được coi là "đã mua thành công":

Ví dụ status code trong hệ thống:
- 0 hoặc "pending" = Chờ xác nhận
- 1 hoặc "confirmed" = Đã xác nhận
- 2 hoặc "shipping" = Đang giao
- 3 hoặc "delivered" = Đã giao
- 4 hoặc "completed" = Hoàn thành
- -1 hoặc "cancelled" = Đã hủy

⚠️ Chỉ tính là "đã mua" khi status là:
   - "delivered" (Đã giao)
   - "completed" (Hoàn thành)

❌ KHÔNG tính các trạng thái:
   - pending, confirmed, shipping (chưa nhận hàng)
   - cancelled (đã hủy)

LƯU Ý QUAN TRỌNG
================================================================

1. PERFORMANCE: Nếu có nhiều comments, nên:
   - Dùng aggregation pipeline (CÁCH 1)
   - HOẶC lưu hasPurchased vào DB khi tạo comment (CÁCH 2)
   - Tránh dùng loop với nhiều query

2. CACHE: Có thể cache kết quả kiểm tra hasPurchased
   - Cache 5-10 phút cho mỗi user-product pair
   - Xóa cache khi có đơn hàng mới hoàn thành

3. INDEX: Đảm bảo có index trên:
   - Order: { customerId: 1, status: 1 }
   - Order: { 'items.productId': 1 }
   - Comment: { productId: 1, createdAt: -1 }

4. SECURITY: 
   - Validate productId là ObjectId hợp lệ
   - Không cho phép fake hasPurchased từ client
